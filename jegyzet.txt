const express = require('express');
const multer  = require('multer');
const crypto = require('crypto');
const path = require('path');

const app = express();

// Configure storage destination & unique filename generator
const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    cb(null, 'uploads/');
  },
  filename: (req, file, cb) => {
    const uniqueName = crypto.randomUUID() + path.extname(file.originalname);
    cb(null, uniqueName);
  }
});

const upload = multer({ storage: storage });

app.post('/tickets/:id/attachments', upload.single('attachment'), async (req, res) => {
  const ticketId = req.params.id;
  const userId = req.user.id; // From auth session/jwt
  const file = req.file;

  // Insert into MariaDB database
  const query = `
    INSERT INTO attachments 
    (ticket_id, uploaded_by, original_filename, stored_filename, file_path, file_size_bytes, mime_type)
    VALUES (?, ?, ?, ?, ?, ?, ?)
  `;
  
  await db.query(query, [
    ticketId, 
    userId, 
    file.originalname, 
    file.filename, 
    file.path, 
    file.size, 
    file.mimetype
  ]);

  res.json({ message: 'File uploaded successfully!' });
});
